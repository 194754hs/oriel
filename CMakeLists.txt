cmake_minimum_required(VERSION 3.21)
project(oriel LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# The 32-bit build produces the shim and nothing else: the shell itself stays
# 64-bit, only the in-proc server has to match whatever process it lands in.
option(ORIEL_SHIM_ONLY "Build only the dialog shim (used for the x86 build)" OFF)

if(NOT ORIEL_SHIM_ONLY)
add_executable(oriel WIN32
    src/main.cpp
    src/app_window.cpp
    src/app_window.h
    src/column_model.cpp
    src/column_model.h
    src/dialog_server.cpp
    src/dialog_server.h
    src/enumerate.cpp

    src/shell_menu.cpp
    src/shell_menu.h
    src/thumbnail.cpp
    src/thumbnail.h
    src/assoc.cpp
    src/assoc.h
    src/actions.cpp
    src/actions.h
    src/tags.cpp
    src/tags.h
    src/textfield.h
    src/settings.cpp
    src/settings.h

    src/i18n.cpp

target_compile_definitions(oriel PRIVATE
    UNICODE _UNICODE
    NOMINMAX
    WIN32_LEAN_AND_MEAN
)

target_compile_options(oriel PRIVATE
    /W4 /permissive- /utf-8 /Zc:__cplusplus
)

target_link_libraries(oriel PRIVATE
    d3d11 dxgi d2d1 dcomp dwrite dwmapi user32 gdi32 shlwapi shell32 ole32
)

# The embedded icon outlines are ISC/MIT, and both licences require their notice
# to travel with the binary. Copying it next to the .exe on every build is the
# only way it cannot be forgotten at packaging time.
add_custom_command(TARGET oriel POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_SOURCE_DIR}/THIRD-PARTY-NOTICES.txt"
            "$<TARGET_FILE_DIR:oriel>/THIRD-PARTY-NOTICES.txt"
    COMMENT "Copying third-party notices next to oriel.exe"
)
endif()  # NOT ORIEL_SHIM_ONLY

# ── the file-dialog shim ────────────────────────────────────────────
# An in-proc COM server that ends up inside other applications, so it stays
# small, has no dependencies beyond the OS, and forwards everything it cannot
# handle to the genuine dialog.
add_library(oriel_dialog SHARED
    shim/shim.cpp
    shim/shim.def
)
target_compile_definitions(oriel_dialog PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(oriel_dialog PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
target_link_libraries(oriel_dialog PRIVATE ole32 shlwapi user32 advapi32)
set_target_properties(oriel_dialog PROPERTIES PREFIX "" OUTPUT_NAME "oriel_dialog")
# Static CRT: this DLL is loaded into other people's processes, so it must not
# depend on a redistributable being present, and must not risk disagreeing with
# whatever runtime the host has already loaded.
set_property(TARGET oriel_dialog PROPERTY
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# ── code signing ────────────────────────────────────────────────────
# Off until a certificate exists. It matters most for the shim, which is loaded
# into other applications' processes: an unsigned DLL turning up inside a signed
# program is exactly what security tooling is built to object to.
#
#   cmake -B build-rel -DORIEL_SIGN_THUMBPRINT=<sha1 of the cert>
set(ORIEL_SIGN_THUMBPRINT "" CACHE STRING "Certificate thumbprint for signtool")
if(ORIEL_SIGN_THUMBPRINT)
    foreach(tgt oriel oriel_dialog)
        if(TARGET ${tgt})
            add_custom_command(TARGET ${tgt} POST_BUILD
                COMMAND signtool sign /sha1 ${ORIEL_SIGN_THUMBPRINT} /fd SHA256
                        /tr http://timestamp.digicert.com /td SHA256
                        "$<TARGET_FILE:${tgt}>"
                COMMENT "Signing ${tgt}")
        endif()
    endforeach()
else()
    message(STATUS "Oriel: unsigned build (set ORIEL_SIGN_THUMBPRINT to sign)")
endif()

# Exercises the shim by loading it directly - no registry involved.
if(NOT ORIEL_SHIM_ONLY)
add_executable(shim_test shim/shim_test.cpp)
target_compile_definitions(shim_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(shim_test PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
target_link_libraries(shim_test PRIVATE ole32 shlwapi user32)
add_dependencies(shim_test oriel_dialog)
endif()
